async function api(path, body) {
  const response = await fetch(path, body === undefined ? {} : {
    method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body)
  })
  const data = await response.json()
  if (!response.ok) {
    const message = data.message + (data.decisionId ? '\nDecision ' + data.decisionId + ' · Policy ' + data.policyVersion : '')
    throw new Error(message)
  }
  return data
}

function options(node, choices) {
  const selected = node.value
  node.replaceChildren(...choices.map(choice => {
    const [value, label] = Array.isArray(choice) ? choice : [choice, choice]
    const option = create('option', label)
    option.value = value
    return option
  }))
  if (Array.from(node.options).some(option => option.value === selected)) node.value = selected
}
function table(headers, rows) {
  const wrapper = create('div')
  wrapper.className = 'scroll'
  const grid = create('table')
  const head = grid.createTHead().insertRow()
  for (const header of headers) {
    const th = create('th', header)
    th.scope = 'col'
    head.append(th)
  }
  const body = grid.createTBody()
  for (const row of rows) {
    const output = body.insertRow()
    for (const value of row) {
      const cell = output.insertCell()
      if (value instanceof Node) cell.append(value)
      else cell.textContent = value ?? ''
    }
  }
  wrapper.append(grid)
  return wrapper
}

function weeklySchedule(slots) {
  const groups = Map.groupBy(slots, slot => slot.resource)
  select('schedule').replaceChildren()
  for (const resourceSlots of groups.values()) {
    const { room, sauna, group } = resourceSlots[0]
    const starts = [...new Set(resourceSlots.map(s => s.start))].sort()
    const rows = starts.map(start => [start, ...weekdays.map(day => {
      const slot = resourceSlots.find(s => s.start === start && s.weekday === day)
      if (!slot) return ''
      const content = create('div')
      const label = slot.end + ' · ' + (slot.state === 'held' ? 'Held by apartment ' + slot.household : 'Free')
        + (slot.timeClass === 'general' ? '' : ' · ' + classLabel(slot.timeClass))
      if (slot.state === 'free') {
        const button = create('button', label)
        button.type = 'button'
        button.className = 'slot'
        button.addEventListener('click', () => fillSlot(slot))
        content.append(button)
      } else {
        content.textContent = label
        content.append(create('div', slot.bookingId))
      }
      return content
    })])
    select('schedule').append(create('h3', sauna + ' · ' + room + ' · Group ' + group), table(['Starts', ...weekdays], rows))
  }
}

function manageFields() {
  const action = manage.elements.action.value
  const inputs = operations[action][1].map(name => {
    const label = create('label', fields[name])
    const input = create(name === 'changes' ? 'textarea' : ['mode', 'decision'].includes(name) ? 'select' : 'input')
    input.name = name
    input.required = action !== 'history'
    if (name === 'date') input.type = 'date'
    if (name === 'household') input.value = book.elements.household.value
    if (name === 'role') input.value = action === 'disputes' ? currentRoute.disputeRole :
      action === 'evaluate-scarcity' ? currentRoute.bookingOperatorRole : book.elements.role.value
    if (name === 'mode') options(input, [['free', 'Free'], ['sale', 'For sale']])
    if (name === 'decision') options(input, [['confirm', 'Confirm'], ['decline', 'Decline']])
    if (name === 'changes') input.placeholder = '{"chooseDays":14,"lapseDays":30}'
    if (name === 'effectiveAt') input.placeholder = '2030-01-01T12:00:00Z'
    label.append(input)
    return label
  })
  select('manage-fields').replaceChildren(...inputs)
}
